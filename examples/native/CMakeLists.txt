cmake_minimum_required(VERSION 3.20)
project(hello_desktop_native LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)

add_library(workbench_native SHARED workbench.c)
target_compile_definitions(workbench_native PRIVATE WB_BUILD)
target_include_directories(workbench_native PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(workbench_probe probe.c)
target_link_libraries(workbench_probe PRIVATE workbench_native)
if(MSVC)
  foreach(target workbench_native workbench_probe)
    target_compile_options(${target} PRIVATE /W4 /WX /Zi)
    target_link_options(${target} PRIVATE /DEBUG)
  endforeach()
else()
  foreach(target workbench_native workbench_probe)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -g)
  endforeach()
endif()
enable_testing()
add_test(NAME native_contract COMMAND workbench_probe)
